    unsigned long hash = 5381;
    int c;
    while ((c = *str++)) {
        hash = ((hash << 5) + hash) + c;
    }
    return hash;
}

        return NULL;

}

    let mut hash: u64 = 5381;
    for c in s.bytes() {
        hash = hash.wrapping_shl(5).wrapping_add(hash).wrapping_add(c as u64);
    }
    hash
}
