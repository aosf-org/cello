/**
 * Simple hash function (djb2)
 */
static unsigned long hash_string(const char* str) {
    unsigned long hash = 5381;
    int c;
    while ((c = *str++)) {
        hash = ((hash << 5) + hash) + c;
    }
    return hash;
}

/// Simple hash map implementation using chaining for collision resolution
pub struct HashMap<T> {
    buckets: Vec<Option<Box<HashMapEntry<T>>>>,
    size: usize,
}

impl<T> HashMap<T> {
    /// Create a new hash map
    pub fn new() -> Self {
        HashMap {
            buckets: vec![None; HASHMAP_INITIAL_CAPACITY],
            size: 0,
        }
    }

    /// Simple hash function (djb2)
    fn hash_string(s: &str) -> u64 {
        let mut hash: u64 = 5381;
        for c in s.bytes() {
            hash = hash.wrapping_shl(5).wrapping_add(hash).wrapping_add(c as u64);
        }
        hash
    }

    /// Get the bucket index for a given key
    fn get_index(&self, key: &str) -> usize {
        let hash = Self::hash_string(key);
        (hash % self.buckets.len() as u64) as usize
    }

    /// Insert or update a key-value pair
    pub fn put(&mut self, key: &str, value: T) -> Result<(), String> {
        let index = self.get_index(key);

        // Check if key exists and update if so
        let mut current = &mut self.buckets[index];
        while let Some(entry) = current {
            if entry.key == key {
                entry.value = value;
                return Ok(());

            key: key.to_string(),
            value,
